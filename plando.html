DB 초기화 SQL (app_users 포함)
외부 Supabase 연동 시 SQL Editor에서 아래 구문을 실행하세요. (미연결 시에도 브라우저 로컬 스토리지에서 모든 회원가입/로그인이 정상 동작합니다.)
-- Enable UUID extension create extension if not exists "pgcrypto"; -- app_users (회원가입 및 로그인 테이블) create table if not exists app_users ( id uuid primary key default gen_random_uuid(), username text not null unique, password_hash text not null, created_at timestamptz default now() ); alter table app_users enable row level security; create policy "public read app_users" on app_users for select using (true); create policy "public insert app_users" on app_users for insert with check (true); -- plans create table if not exists plans ( id uuid primary key default gen_random_uuid(), title text not null, period_start date, period_end date, priority text default 'medium', success_criteria text, estimated_minutes integer default 0, notes text, created_at timestamptz default now() ); alter table plans enable row level security; create policy "public read plans" on plans for select using (true); create policy "public insert plans" on plans for insert with check (true); create policy "public update plans" on plans for update using (true); create policy "public delete plans" on plans for delete using (true); -- plan_versions create table if not exists plan_versions ( id uuid primary key default gen_random_uuid(), plan_id uuid references plans(id) on delete cascade, version_num integer, title text, period_start date, period_end date, priority text, success_criteria text, estimated_minutes integer default 0, notes text, saved_at timestamptz default now() ); alter table plan_versions enable row level security; create policy "public read plan_versions" on plan_versions for select using (true); create policy "public insert plan_versions" on plan_versions for insert with check (true); create policy "public update plan_versions" on plan_versions for update using (true); create policy "public delete plan_versions" on plan_versions for delete using (true); -- tasks create table if not exists tasks ( id uuid primary key default gen_random_uuid(), plan_id uuid references plans(id) on delete set null, title text not null, status text default 'todo', due_date date, priority text default 'medium', tags text, estimated_minutes integer default 0, notes text, completed_at timestamptz, created_at timestamptz default now() ); alter table tasks enable row level security; create policy "public read tasks" on tasks for select using (true); create policy "public insert tasks" on tasks for insert with check (true); create policy "public update tasks" on tasks for update using (true); create policy "public delete tasks" on tasks for delete using (true); -- executions create table if not exists executions ( id uuid primary key default gen_random_uuid(), task_id uuid references tasks(id) on delete cascade, start_at timestamptz, end_at timestamptz, actual_minutes integer default 0, blocker_reason text default '', created_at timestamptz default now() ); alter table executions enable row level security; create policy "public read executions" on executions for select using (true); create policy "public insert executions" on executions for insert with check (true); create policy "public update executions" on executions for update using (true); create policy "public delete executions" on executions for delete using (true); -- sees create table if not exists sees ( id uuid primary key default gen_random_uuid(), date date not null unique, good text, bad text, next_plan text, carry_forward text, rating integer default 0, updated_at timestamptz default now() ); alter table sees enable row level security; create policy "public read sees" on sees for select using (true); create policy "public insert sees" on sees for insert with check (true); create policy "public update sees" on sees for update using (true); create policy "public delete sees" on sees for delete using (true);
